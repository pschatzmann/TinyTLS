99 f.mul(P.Y, P.Y, gamma);
100 f.mul(P.X, gamma, beta);
101 f.subMod(P.X, delta, t1);
102 f.addMod(P.X, delta, t2);
103 f.mul(t1, t2, alpha);
104 f.addMod(alpha, alpha, t1);
105 f.addMod(alpha, t1, alpha); // alpha = 3 (X - delta)(X + delta)
106 f.mul(alpha, alpha, t1);

113 f.subMod(t1, gamma, t1);

197 bool onCurve(const BigInt &x, const BigInt &y) const {
198 const Montgomery &f = fp_;
199 BigInt xm = f.enter(x), ym = f.enter(y), lhs, rhs, t;
200 f.mul(ym, ym, lhs);
201 f.mul(xm, xm, rhs);
202 f.mul(rhs, xm, rhs);
203 f.addMod(xm, xm, t);
204 f.addMod(t, xm, t);
205 f.subMod(rhs, t, rhs);
206 f.addMod(rhs, bm_, rhs);
207 return lhs.compare(rhs) == 0;
208 }

216 Error verifyWith(const uint8_t qx_b[32], const uint8_t qy_b[32], const uint8_t *digest,
217 size_t digest_len, const uint8_t r_b[32], const uint8_t s_b[32]) const {
218 const Montgomery &fn = fn_;
219 std::unique_ptr<Workspace> ws(new Workspace());
220 BigInt &qx = ws->qx, &qy = ws->qy, &r = ws->r, &s = ws->s, &e = ws->e, &two = ws->two,
221 &nm2 = ws->nm2, &w = ws->w, &t = ws->t, &u1 = ws->u1, &u2 = ws->u2, &pm2 = ws->pm2,
222 &zn = ws->zn, &zinv = ws->zinv, &x = ws->x;
223 Point &Q = ws->Q, &G1 = ws->G1, &G2 = ws->G2, &R = ws->R;
224
225 qx.fromBytes(qx_b, 32);
226 qy.fromBytes(qy_b, 32);
227 if (qx.compare(fp_.modulus()) >= 0 || qy.compare(fp_.modulus()) >= 0 || !onCurve(qx, qy)) {
228 return Error::EcdsaBadInput;
229 }
230 r.fromBytes(r_b, 32);
231 s.fromBytes(s_b, 32);
232 if (r.isZero() || s.isZero() || r.compare(fn.modulus()) >= 0 || s.compare(fn.modulus()) >= 0) {
233 return Error::EcdsaVerifyFailed;
234 }
235
236 // e = leftmost 256 bits of the digest, reduced mod n.
237 e.fromBytes(digest, digest_len < 32 ? digest_len : 32);
238 if (e.compare(fn.modulus()) >= 0) e.sub(fn.modulus(), e);
239
240 // w = s^-1 mod n (n is prime), u1 = e w, u2 = r w.
241 two = BigInt(2);
242 fn.modulus().sub(two, nm2);
243 w = fn.powMont(fn.enter(s), nm2);
244 fn.mul(fn.enter(r), w, t);
245 u2 = fn.leave(t);
246 fn.mul(fn.enter(e), w, t);
247 u1 = fn.leave(t);
248
249 // R = u1 G + u2 Q
250 Q.X = fp_.enter(qx);
251 Q.Y = fp_.enter(qy);
252 Q.Z = fp_.one();
253 mul(u1, g_, G1);
254 mul(u2, Q, G2);
255 add(G1, G2, R);
256 if (R.Z.isZero()) return Error::EcdsaVerifyFailed;
257
258 // x = X / Z^2 (affine), reduced mod n, must equal r.
259 fp_.modulus().sub(two, pm2);
260 zn = fp_.leave(R.Z);
261 zinv = fp_.pow(zn, pm2);
262 zinv = fp_.enter(zinv);
263 fp_.mul(zinv, zinv, zinv);
264 fp_.mul(R.X, zinv, x);
265 x = fp_.leave(x);
266 if (x.compare(fn.modulus()) >= 0) x.sub(fn.modulus(), x);
267 return x.compare(r) == 0 ? Error::Ok : Error::EcdsaVerifyFailed;
268 }
