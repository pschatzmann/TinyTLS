42 void start(const uint8_t *key, size_t key_len) {
43 // The padded key is built in opad_ first, then turned into the outer and inner pads.
44 memset(opad_, 0, sizeof(opad_));
45 if (key_len > Sha256::BlockSize) {
46 Sha256 h;
47 h.digest(key, key_len, opad_); // the 32-byte digest; the rest stays zero
48 } else if (key_len) {
49 memcpy(opad_, key, key_len);
50 }
51 for (size_t i = 0; i < Sha256::BlockSize; i++) {
52 uint8_t k = opad_[i];
53 opad_[i] = k ^ 0x5c;
54 pad_[i] = k ^ 0x36;
55 }
56 inner_.reset();
57 inner_.update(pad_, sizeof(pad_));
58 memset(pad_, 0, sizeof(pad_));
59 }
60
62 void update(const uint8_t *data, size_t len) { inner_.update(data, len); }
63
65 void finish(uint8_t tag[TagSize]) {
66 uint8_t inner_hash[32];
67 inner_.finish(inner_hash);
68 Sha256 outer;
69 outer.update(opad_, sizeof(opad_));
70 outer.update(inner_hash, sizeof(inner_hash));
71 outer.finish(tag);
72 }
73
75 void compute(const uint8_t *key, size_t key_len, const uint8_t *data, size_t len,
76 uint8_t tag[TagSize]) {
77 start(key, key_len);
78 update(data, len);
79 finish(tag);
80 }
81
82 private:
83 Sha256 inner_;
84 uint8_t opad_[Sha256::BlockSize];
85 uint8_t pad_[Sha256::BlockSize];
86};

123 bool expand(const uint8_t prk[32], const uint8_t *info, size_t info_len, uint8_t *out,
124 size_t out_len) {
125 if (out_len > 255 * 32) return false;
126 uint8_t t[32];
127 size_t t_len = 0, done = 0;
128 uint8_t counter = 1;
129 while (done < out_len) {
130 hmac_.start(prk, 32);
131 hmac_.update(t, t_len);
132 hmac_.update(info, info_len);
133 hmac_.update(&counter, 1);
134 hmac_.finish(t);
135 t_len = 32;
136 size_t n = out_len - done < 32 ? out_len - done : 32;
137 memcpy(out + done, t, n);
138 done += n;
139 counter++;
140 }
141 memset(t, 0, sizeof(t));
142 return true;
143 }

146 bool expandLabel(const uint8_t secret[32], const char *label, const uint8_t *context,
147 size_t context_len, uint8_t *out, size_t out_len) {
148 const char prefix[] = "tls13 ";
149 size_t label_len = strlen(label);
150 size_t full_label_len = sizeof(prefix) - 1 + label_len;
151 if (full_label_len > 255 || context_len > 255 || out_len > 0xFFFF) return false;
152
153 uint8_t *info = info_.data();
154 size_t p = 0;
155 info[p++] = (uint8_t)(out_len >> 8);
156 info[p++] = (uint8_t)out_len;
157 info[p++] = (uint8_t)full_label_len;
158 memcpy(info + p, prefix, sizeof(prefix) - 1);
159 p += sizeof(prefix) - 1;
160 memcpy(info + p, label, label_len);
161 p += label_len;
162 info[p++] = (uint8_t)context_len;
163 if (context_len) memcpy(info + p, context, context_len);
164 p += context_len;
165
166 uint8_t prk[32];
167 memcpy(prk, secret, 32);
168 bool rc = expand(prk, info, p, out, out_len);
169 memset(prk, 0, sizeof(prk));
170 return rc;
171 }
