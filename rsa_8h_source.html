57 Error verifyPkcs1Sha256(const uint8_t hash[32], const uint8_t *sig, size_t sig_len) const {
58 static const uint8_t digest_info[19] = {0x30, 0x31, 0x30, 0x0d, 0x06, 0x09, 0x60, 0x86, 0x48, 0x01,
59 0x65, 0x03, 0x04, 0x02, 0x01, 0x05, 0x00, 0x04, 0x20};
60 size_t k = n_len_;
61 if (k < 11 + 19 + 32) return Error::RsaVerifyFailed;
62 HeapArray<uint8_t> em_store(MaxModulusBytes);
63 uint8_t *em = em_store.data();
64 if (publicOp(sig, sig_len, em) != Error::Ok) return Error::RsaVerifyFailed;
65
66 // EM = 00 01 FF..FF 00 DigestInfo H
67 const size_t t_len = 19 + 32;
68 uint8_t expected[19 + 32];
69 memcpy(expected, digest_info, 19);
70 memcpy(expected + 19, hash, 32);
71 if (em[0] != 0x00 || em[1] != 0x01) return Error::RsaVerifyFailed;
72 size_t sep = k - t_len - 1; // index of the 0x00 separator
73 for (size_t i = 2; i < sep; i++) {
74 if (em[i] != 0xff) return Error::RsaVerifyFailed;
75 }
76 if (sep < 10 || em[sep] != 0x00) return Error::RsaVerifyFailed;
77 uint8_t diff = 0;
78 for (size_t i = 0; i < t_len; i++) diff |= (uint8_t)(em[sep + 1 + i] ^ expected[i]);
79 return diff == 0 ? Error::Ok : Error::RsaVerifyFailed;

83 Error verifyPssSha256(const uint8_t hash[32], const uint8_t *sig, size_t sig_len) const {
84 const size_t h_len = 32, s_len = 32;
85 BigInt n;
86 if (!n.fromBytes(n_, n_len_)) return Error::RsaVerifyFailed;
87 size_t em_bits = n.bitLength() - 1;
88 size_t em_len = (em_bits + 7) / 8;
89 size_t k = n_len_;
90 if (em_len < h_len + s_len + 2 || k < em_len) return Error::RsaVerifyFailed;
91
92 HeapArray<uint8_t> full_store(MaxModulusBytes);
93 uint8_t *full = full_store.data();
94 if (publicOp(sig, sig_len, full) != Error::Ok) return Error::RsaVerifyFailed;
95 for (size_t i = 0; i < k - em_len; i++) {
96 if (full[i] != 0) return Error::RsaVerifyFailed;
97 }
98 const uint8_t *em = full + (k - em_len);
99 if (em[em_len - 1] != 0xbc) return Error::RsaVerifyFailed;
100
101 size_t db_len = em_len - h_len - 1;
102 const uint8_t *masked_db = em;
103 const uint8_t *h = em + db_len;
104 HeapArray<uint8_t> db_store(MaxModulusBytes);
105 uint8_t *db = db_store.data();
106 mgf1Sha256(h, h_len, db, db_len);
107 for (size_t i = 0; i < db_len; i++) db[i] ^= masked_db[i];
108 size_t top_bits = 8 * em_len - em_bits;
109 db[0] &= (uint8_t)(0xff >> top_bits);
110
111 size_t ps_len = db_len - s_len - 1;
112 for (size_t i = 0; i < ps_len; i++) {
113 if (db[i] != 0) return Error::RsaVerifyFailed;
114 }
115 if (db[ps_len] != 0x01) return Error::RsaVerifyFailed;
116 const uint8_t *salt = db + ps_len + 1;
117
118 // H' = Hash(8 zero bytes || mHash || salt)
119 Sha256 ctx;
120 uint8_t zeros[8] = {0};
121 ctx.update(zeros, 8);
122 ctx.update(hash, 32);
123 ctx.update(salt, s_len);
124 uint8_t h_prime[32];
125 ctx.finish(h_prime);
126
127 uint8_t diff = 0;
128 for (size_t i = 0; i < h_len; i++) diff |= (uint8_t)(h[i] ^ h_prime[i]);
129 return diff == 0 ? Error::Ok : Error::RsaVerifyFailed;
130 }

134 Error publicOp(const uint8_t *sig, size_t sig_len, uint8_t *out) const {
135 BigInt n, s, e, m;
136 if (!n.fromBytes(n_, n_len_) || !e.fromBytes(e_, e_len_)) return Error::BadInput;
137 if (sig_len != n_len_) return Error::RsaVerifyFailed;
138 if (!s.fromBytes(sig, sig_len)) return Error::RsaVerifyFailed;
139 Montgomery ctx;
140 if (!ctx.init(n)) return Error::BadInput;
141 if (s.compare(ctx.modulus()) >= 0) return Error::RsaVerifyFailed;
142 m = ctx.pow(s, e);
143 return m.toBytes(out, n_len_) ? Error::Ok : Error::RsaVerifyFailed;
144 }

147 void mgf1Sha256(const uint8_t *seed, size_t seed_len, uint8_t *mask, size_t mask_len) const {
148 size_t done = 0;
149 uint32_t counter = 0;
150 while (done < mask_len) {
151 uint8_t c[4] = {(uint8_t)(counter >> 24), (uint8_t)(counter >> 16), (uint8_t)(counter >> 8),
152 (uint8_t)counter};
153 Sha256 h;
154 h.update(seed, seed_len);
155 h.update(c, 4);
156 uint8_t block[32];
157 h.finish(block);
158 size_t n = mask_len - done < 32 ? mask_len - done : 32;
159 memcpy(mask + done, block, n);
160 done += n;
161 counter++;
162 }
163 }
