386inline Error Session::sendRecord(uint8_t type, const uint8_t *data, size_t len) {
387 if (tx_pending_) return Error::Internal;
388 uint8_t *out = tx_buf_.data();
389 size_t body;
390 if (tx_keys_.active()) {
391 if (len > kMaxFragment) return Error::BadInput;
392 body = len + 1 + 16; // content, inner type, tag
393 out[0] = kTypeApplication; // the outer type of protected records is always 23
394 out[1] = 0x03;
395 out[2] = 0x03;
396 out[3] = (uint8_t)(body >> 8);
397 out[4] = (uint8_t)body;
398 if (len) memcpy(out + 5, data, len);
399 out[5 + len] = type;
400 uint8_t tag[16];
401 if (!tx_keys_.seal(out, kHeaderSize, out + 5, len + 1, tag)) return Error::Internal;
402 memcpy(out + 5 + len + 1, tag, 16);
403 tx_keys_.advance();
404 } else {
405 if (len > kMaxFragment) return Error::BadInput;
406 body = len;
407 out[0] = type;
408 out[1] = 0x03;
409 out[2] = 0x01; // legacy_record_version of the initial ClientHello
410 out[3] = (uint8_t)(body >> 8);
411 out[4] = (uint8_t)body;
412 if (len) memcpy(out + 5, data, len);
413 }
414 tx_len_ = kHeaderSize + body;
415 tx_off_ = 0;
416 tx_pending_ = true;
417 return Error::Ok;
418}

443inline Error Session::fetchRecord() {
444 if (peer_close_notify_) return Error::PeerCloseNotify;
445 if (peer_alert_) return Error::AlertReceived;
446 for (;;) {
447 Error rc;
448 size_t got = 0;
449 while (rx_hdr_have_ < kHeaderSize) {
450 rc = recvSome(rx_hdr_ + rx_hdr_have_, kHeaderSize - rx_hdr_have_, got);
451 if (rc != Error::Ok) return rc;
452 rx_hdr_have_ += got;
453 }
454 uint8_t type = rx_hdr_[0];
455 size_t len = rd16(rx_hdr_ + 3);
456 if (rx_hdr_[1] != 0x03) return Error::InvalidRecord;
457 if (type < kTypeChangeCipherSpec || type > kTypeApplication) return Error::InvalidRecord;
458 size_t max_len = rx_keys_.active() ? kMaxCiphertext : kMaxFragment;
459 if (len > max_len) return Error::InvalidRecord;
460 while (rx_have_ < len) {
461 rc = recvSome(rx_buf_.data() + rx_have_, len - rx_have_, got);
462 if (rc != Error::Ok) return rc;
463 rx_have_ += got;
464 }
465
466 // ChangeCipherSpec is never encrypted, even after keys are installed.
467 if (rx_keys_.active() && type != kTypeChangeCipherSpec) {
468 if (type != kTypeApplication || len < 17) return Error::InvalidRecord;
469 size_t ct = len - 16;
470 if (rx_keys_.open(rx_hdr_, kHeaderSize, rx_buf_.data(), ct, rx_buf_.data() + ct) != Error::Ok) {
471 return Error::InvalidRecord; // bad record MAC
472 }
473 rx_keys_.advance();
474 size_t n = ct;
475 while (n > 0 && rx_buf_[n - 1] == 0) n--; // strip zero padding
476 if (n == 0) return Error::InvalidRecord;
477 pt_type_ = rx_buf_[n - 1];
478 pt_len_ = n - 1;
479 } else {
480 pt_type_ = type;
481 pt_len_ = len;
482 }
483 pt_off_ = 0;
484 rx_hdr_have_ = 0;
485 rx_have_ = 0;
486
487 if (pt_type_ == kTypeChangeCipherSpec) {
488 if (pt_len_ != 1 || rx_buf_[0] != 1) return Error::InvalidRecord;
489 pt_len_ = 0;
490 continue;
491 }
492 if (pt_type_ == kTypeAlert) {
493 if (pt_len_ < 2) return Error::InvalidRecord;
494 uint8_t level = rx_buf_[0];
495 uint8_t desc = rx_buf_[1];
496 pt_len_ = 0;
497 if (desc == 0) { // close_notify
498 peer_close_notify_ = true;
499 return Error::PeerCloseNotify;
500 }
501 if (level == 2) { // fatal
502 peer_alert_ = desc ? desc : 1;
503 return Error::AlertReceived;
504 }
505 continue; // warnings are ignored
506 }
507 if (pt_len_ == 0) continue; // empty fragment
508 return Error::Ok;
509 }

512inline Error Session::readHandshake(const uint8_t *&msg, size_t &msg_len) {
513 for (;;) {
514 if (hs_need_ && hs_have_ == hs_need_) break;
515 if (pt_len_ == 0) {
516 Error rc = fetchRecord();
517 if (rc != Error::Ok) return rc;
518 }
519 if (pt_type_ != kTypeHandshake) return Error::UnexpectedMessage;
520 size_t want = hs_have_ < 4 ? 4 - hs_have_ : hs_need_ - hs_have_;
521 size_t n = want < pt_len_ ? want : pt_len_;
522 memcpy(hs_buf_.data() + hs_have_, rx_buf_.data() + pt_off_, n);
523 hs_have_ += n;
524 pt_off_ += n;
525 pt_len_ -= n;
526 if (hs_need_ == 0 && hs_have_ == 4) {
527 size_t body = rd24(hs_buf_.data() + 1);
528 if (4 + body > kMaxHandshake) return Error::MessageTooLarge;
529 hs_need_ = 4 + body;
530 }
531 }
532 msg = hs_buf_.data();
533 msg_len = hs_need_;
534 return Error::Ok;
535}

537inline Error Session::buildClientHello() {
538 if (rng_ == nullptr) return Error::BadInput;
539 uint8_t rnd[32 + 32 + 32];
540 if (!rng_->random(rnd, sizeof(rnd))) return Error::EntropyFailed;
541 memcpy(client_random_, rnd, 32);
542 memcpy(session_id_, rnd + 32, 32);
543 memcpy(x_priv_, rnd + 64, 32);
544 memset(rnd, 0, sizeof(rnd));
545 X25519 x25519;
546 if (!x25519.publicKey(x_priv_, x_pub_)) return Error::X25519LowOrder;
547
548 HeapArray<uint8_t> m_store(512);
549 uint8_t *m = m_store.data();
550 size_t o = 4; // handshake header is filled in at the end
551 o = wr16(m, o, 0x0303);
552 memcpy(m + o, client_random_, 32);
553 o += 32;
554 m[o++] = 32;
555 memcpy(m + o, session_id_, 32);
556 o += 32;
557 o = wr16(m, o, 4);
558 o = wr16(m, o, TrafficKeys::CipherAes128GcmSha256);
559 o = wr16(m, o, TrafficKeys::CipherChaCha20Poly1305Sha256);
560 m[o++] = 1; // legacy_compression_methods
561 m[o++] = 0;
562
563 size_t ext_len_pos = o;
564 o += 2;
565 size_t ext_start = o;
566 if (has_hostname_) {
567 size_t hl = strlen(hostname_);
568 o = wr16(m, o, 0x0000); // server_name
569 o = wr16(m, o, hl + 5);
570 o = wr16(m, o, hl + 3);
571 m[o++] = 0; // host_name
572 o = wr16(m, o, hl);
573 memcpy(m + o, hostname_, hl);
574 o += hl;
575 }
576 o = wr16(m, o, 0x000a); // supported_groups
577 o = wr16(m, o, 4);
578 o = wr16(m, o, 2);
579 o = wr16(m, o, 0x001d); // x25519
580 o = wr16(m, o, 0x000d); // signature_algorithms
581 o = wr16(m, o, 8);
582 o = wr16(m, o, 6);
583 o = wr16(m, o, 0x0804); // rsa_pss_rsae_sha256
584 o = wr16(m, o, 0x0403); // ecdsa_secp256r1_sha256
585 o = wr16(m, o, 0x0401); // rsa_pkcs1_sha256
586 o = wr16(m, o, 0x002b); // supported_versions
587 o = wr16(m, o, 3);
588 m[o++] = 2;
589 o = wr16(m, o, 0x0304);
590 o = wr16(m, o, 0x0033); // key_share
591 o = wr16(m, o, 38);
592 o = wr16(m, o, 36);
593 o = wr16(m, o, 0x001d);
594 o = wr16(m, o, 32);
595 memcpy(m + o, x_pub_, 32);
596 o += 32;
597 wr16(m, ext_len_pos, (uint32_t)(o - ext_start));
598
599 m[0] = kHsClientHello;
600 m[1] = (uint8_t)((o - 4) >> 16);
601 m[2] = (uint8_t)((o - 4) >> 8);
602 m[3] = (uint8_t)(o - 4);
603 transcript_.update(m, o);
604 Error rc = sendRecord(kTypeHandshake, m, o);
605 m_store.wipe();
606 return rc;
607}

609inline Error Session::processServerHello(const uint8_t *m, size_t mlen) {
610 static const uint8_t hrr_random[32] = {
611 0xCF, 0x21, 0xAD, 0x74, 0xE5, 0x9A, 0x61, 0x11, 0xBE, 0x1D, 0x8C, 0x02, 0x1E, 0x65, 0xB8, 0x91,
612 0xC2, 0xA2, 0x11, 0x16, 0x7A, 0xBB, 0x8C, 0x5E, 0x07, 0x9E, 0x09, 0xE2, 0xC8, 0xA8, 0x33, 0x9C};
613 const uint8_t *b = m + 4;
614 size_t bl = mlen - 4;
615 if (bl < 2 + 32 + 1) return Error::BadServerHello;
616 if (b[0] != 0x03 || b[1] != 0x03) return Error::BadServerHello;
617 if (memcmp(b + 2, hrr_random, 32) == 0) return Error::Unsupported; // HelloRetryRequest
618
619 size_t o = 34;
620 size_t sid_len = b[o++];
621 if (sid_len != 32 || o + sid_len + 2 + 1 + 2 > bl) return Error::BadServerHello;
622 if (memcmp(b + o, session_id_, 32) != 0) return Error::BadServerHello;
623 o += 32;
624 uint16_t cipher = rd16(b + o);
625 o += 2;
626 if (b[o] != 0) return Error::BadServerHello;
627 o += 1;
628 size_t ext_len = rd16(b + o);
629 o += 2;
630 if (o + ext_len != bl) return Error::BadServerHello;
631
632 bool version_ok = false, share_ok = false;
633 uint8_t server_pub[32];
634 size_t end = o + ext_len;
635 while (o < end) {
636 if (o + 4 > end) return Error::BadServerHello;
637 uint16_t et = rd16(b + o);
638 size_t el = rd16(b + o + 2);
639 o += 4;
640 if (o + el > end) return Error::BadServerHello;
641 const uint8_t *d = b + o;
642 if (et == 0x002b) { // supported_versions
643 if (el != 2 || rd16(d) != 0x0304) return Error::ProtocolVersion;
644 version_ok = true;
645 } else if (et == 0x0033) { // key_share
646 if (el != 36 || rd16(d) != 0x001d || rd16(d + 2) != 32) return Error::BadServerHello;
647 memcpy(server_pub, d + 4, 32);
648 share_ok = true;
649 }
650 o += el;
651 }
652 if (!version_ok || !share_ok) return Error::BadServerHello;
653 if (cipher != TrafficKeys::CipherAes128GcmSha256 &&
654 cipher != TrafficKeys::CipherChaCha20Poly1305Sha256) {
655 return Error::BadServerHello;
656 }
657 cipher_ = cipher;
658
659 // Key-schedule scratch (see ks_): th | empty | zeros | early | derived | c_hs | s_hs | shared.
660 uint8_t *ks = ks_.data();
661 uint8_t *th = ks, *empty = ks + 32, *zeros = ks + 64, *early = ks + 96, *derived = ks + 128,
662 *c_hs = ks + 160, *s_hs = ks + 192, *shared = ks + 224;
663 memset(zeros, 0, 32);
664 X25519 x25519;
665 if (!x25519.scalarmult(shared, x_priv_, server_pub)) return Error::BadServerHello;
666
667 // Key schedule (RFC 8446 section 7.1) without PSK.
668 transcript_.update(m, mlen);
669 transcriptHash(th);
670 memcpy(empty, kEmptySha256, 32);
671 hkdf_.extract(nullptr, 0, zeros, 32, early);
672 hkdf_.deriveSecret(early, "derived", empty, derived);
673 hkdf_.extract(derived, 32, shared, 32, handshake_secret_);
674 hkdf_.deriveSecret(handshake_secret_, "c hs traffic", th, c_hs);
675 hkdf_.deriveSecret(handshake_secret_, "s hs traffic", th, s_hs);
676 hkdf_.expandLabel(c_hs, "finished", nullptr, 0, finished_key_c_, 32);
677 hkdf_.expandLabel(s_hs, "finished", nullptr, 0, finished_key_s_, 32);
678 Error rc = tx_keys_.install(cipher, c_hs) ? Error::Ok : Error::Internal;
679 if (rc == Error::Ok) rc = rx_keys_.install(cipher, s_hs) ? Error::Ok : Error::Internal;
680 ks_.wipe();
681 return rc;
682}

684inline Error Session::processCertificate(const uint8_t *m, size_t mlen) {
685 const uint8_t *b = m + 4;
686 size_t bl = mlen - 4;
687 if (bl < 4 || b[0] != 0) return Error::BadCertificate;
688 size_t list_len = rd24(b + 1);
689 if (4 + list_len != bl || list_len == 0) return Error::BadCertificate;
690
691 // Parse the chain in place: the DER pointers stay valid until the message is consumed.
692 size_t o = 4;
693 size_t n = 0;
694 bool parse_failed = false;
695 while (o < bl) {
696 if (o + 3 > bl) return Error::BadCertificate;
697 size_t clen = rd24(b + o);
698 o += 3;
699 if (clen == 0 || o + clen + 2 > bl) return Error::BadCertificate;
700 const uint8_t *der = b + o;
701 o += clen;
702 size_t elen = rd16(b + o);
703 o += 2;
704 if (o + elen > bl) return Error::BadCertificate;
705 o += elen;
706 if (n < TINYTLS_X509_MAX_CHAIN) {
707 if (chain_[n].parse(der, clen) != Error::Ok) parse_failed = true;
708 } else {
709 parse_failed = true; // longer than TINYTLS_X509_MAX_CHAIN
710 }
711 n++;
712 }
713 if (o != bl || n == 0) return Error::BadCertificate;
714
715 if (mode_ == VerifyMode::None) {
716 verify_flags_ |= VerifySkipped;
717 if (!parse_failed) leaf_key_ = chain_[0].publicKey();
718 return Error::Ok;
719 }
720
721 uint32_t flags = 0;
722 Error rc;
723 if (parse_failed) {
724 flags |= VerifyBadCertificate;
725 rc = Error::X509BadCertificate;
726 } else {
727 int64_t now = time_ ? time_() : -1;
728 ChainVerifier verifier(trust_, hostname_, now);
729 rc = verifier.verify(chain_.data(), n, flags);
730 leaf_key_ = chain_[0].publicKey();
731 }
732 verify_flags_ |= flags;
733 if (mode_ == VerifyMode::Required && (rc != Error::Ok || flags != 0)) {
734 return Error::CertificateVerifyFailed;
735 }
736 return Error::Ok;
737}

739inline Error Session::verifyCertificateVerify(const uint8_t *m, size_t mlen) {
740 const uint8_t *b = m + 4;
741 size_t bl = mlen - 4;
742 if (bl < 4) return Error::BadCertificate;
743 uint16_t scheme = rd16(b);
744 size_t sig_len = rd16(b + 2);
745 if (4 + sig_len != bl) return Error::BadCertificate;
746 if (mode_ == VerifyMode::None) return Error::Ok;
747
748 // Signed content: 64 spaces, context string, a zero byte, transcript hash (RFC 8446 section 4.4.3).
749 static const char context[] = "TLS 1.3, server CertificateVerify";
750 const size_t ctx_len = sizeof(context) - 1;
751 uint8_t content[64 + sizeof(context) - 1 + 1 + 32];
752 memset(content, 0x20, 64);
753 memcpy(content + 64, context, ctx_len);
754 content[64 + ctx_len] = 0;
755 transcriptHash(content + 64 + ctx_len + 1);
756
757 if (leaf_key_.verify(scheme, content, sizeof(content), b + 4, sig_len) == Error::Ok) {
758 return Error::Ok;
759 }
760 verify_flags_ |= VerifyBadSignature;
761 return mode_ == VerifyMode::Required ? Error::X509BadSignature : Error::Ok;
762}

764inline Error Session::deriveApplicationKeys(const uint8_t th_sf[32]) {
765 // Key-schedule scratch (see ks_): empty | zeros | derived2 | master | c_ap | s_ap.
766 uint8_t *ks = ks_.data();
767 uint8_t *empty = ks, *zeros = ks + 32, *derived2 = ks + 64, *master = ks + 96, *c_ap = ks + 128,
768 *s_ap = ks + 160;
769 memcpy(empty, kEmptySha256, 32);
770 memset(zeros, 0, 32);
771 hkdf_.deriveSecret(handshake_secret_, "derived", empty, derived2);
772 hkdf_.extract(derived2, 32, zeros, 32, master);
773 hkdf_.deriveSecret(master, "c ap traffic", th_sf, c_ap);
774 hkdf_.deriveSecret(master, "s ap traffic", th_sf, s_ap);
775
776 // Client Finished is still sent under the client handshake keys.
777 uint8_t verify[32];
778 hmac_.start(finished_key_c_, 32);
779 hmac_.update(th_sf, 32);
780 hmac_.finish(verify);
781 uint8_t fin[4 + 32];
782 fin[0] = kHsFinished;
783 fin[1] = 0;
784 fin[2] = 0;
785 fin[3] = 32;
786 memcpy(fin + 4, verify, 32);
787 Error rc = sendRecord(kTypeHandshake, fin, sizeof(fin));
788 if (rc == Error::Ok) {
789 tx_keys_.install(cipher_, c_ap);
790 rx_keys_.install(cipher_, s_ap);
791 }
792 ks_.wipe();
793 return rc;
794}

796inline Error Session::handshake() {
797 if (net_ == nullptr || rng_ == nullptr) return Error::BadInput;
798 if (handshake_complete_) return Error::Ok;
799
800 for (;;) {
801 Error rc = Error::Ok;
802 const uint8_t *msg = nullptr;
803 size_t mlen = 0;
804 switch (state_) {
805 case State::Start:
806 rc = buildClientHello();
807 if (rc != Error::Ok) return rc;
808 state_ = State::FlushClientHello;
809 break;
810
811 case State::FlushClientHello:
812 rc = flush();
813 if (rc != Error::Ok) return rc;
814 state_ = State::ServerHello;
815 break;
816
817 case State::ServerHello:
818 rc = readHandshake(msg, mlen);
819 if (rc != Error::Ok) return rc;
820 if (msg[0] != kHsServerHello) return Error::UnexpectedMessage;
821 rc = processServerHello(msg, mlen);
822 if (rc != Error::Ok) return rc;
823 // Plaintext bytes after ServerHello are not allowed.
824 if (rx_keys_.active() && pt_len_ != 0) return Error::UnexpectedMessage;
825 consumeHandshake();
826 state_ = State::EncryptedExtensions;
827 break;
828
829 case State::EncryptedExtensions:
830 rc = readHandshake(msg, mlen);
831 if (rc != Error::Ok) return rc;
832 if (msg[0] != kHsEncryptedExtensions) return Error::UnexpectedMessage;
833 if (mlen < 4 + 2 || rd16(msg + 4) != mlen - 6) return Error::BadInput;
834 transcript_.update(msg, mlen);
835 consumeHandshake();
836 state_ = State::Certificate;
837 break;
838
839 case State::Certificate:
840 rc = readHandshake(msg, mlen);
841 if (rc != Error::Ok) return rc;
842 if (msg[0] == kHsCertificateRequest) return Error::Unsupported; // client certificates
843 if (msg[0] != kHsCertificate) return Error::UnexpectedMessage;
844 rc = processCertificate(msg, mlen);
845 if (rc != Error::Ok) return rc;
846 transcript_.update(msg, mlen);
847 consumeHandshake();
848 state_ = State::CertificateVerify;
849 break;
850
851 case State::CertificateVerify:
852 rc = readHandshake(msg, mlen);
853 if (rc != Error::Ok) return rc;
854 if (msg[0] != kHsCertificateVerify) return Error::UnexpectedMessage;
855 rc = verifyCertificateVerify(msg, mlen);
856 if (rc != Error::Ok) return rc;
857 transcript_.update(msg, mlen);
858 consumeHandshake();
859 state_ = State::Finished;
860 break;
861
862 case State::Finished: {
863 rc = readHandshake(msg, mlen);
864 if (rc != Error::Ok) return rc;
865 if (msg[0] != kHsFinished || mlen != 4 + 32) return Error::BadFinished;
866 uint8_t th[32], expected[32];
867 transcriptHash(th);
868 hmac_.start(finished_key_s_, 32);
869 hmac_.update(th, 32);
870 hmac_.finish(expected);
871 uint8_t diff = 0;
872 for (int i = 0; i < 32; i++) diff |= (uint8_t)(expected[i] ^ msg[4 + i]);
873 if (diff != 0) return Error::BadFinished;
874 transcript_.update(msg, mlen);
875 consumeHandshake();
876 uint8_t th_sf[32];
877 transcriptHash(th_sf);
878 rc = deriveApplicationKeys(th_sf);
879 if (rc != Error::Ok) return rc;
880 state_ = State::FlushClientFinished;
881 break;
882 }
883
884 case State::FlushClientFinished:
885 rc = flush();
886 if (rc != Error::Ok) return rc;
887 handshake_complete_ = true;
888 state_ = State::Done;
889 memset(handshake_secret_, 0, sizeof(handshake_secret_));
890 memset(finished_key_c_, 0, sizeof(finished_key_c_));
891 memset(finished_key_s_, 0, sizeof(finished_key_s_));
892 return Error::Ok;
893
894 case State::Done:
895 return Error::Ok;
896 }
897 }
898}

900inline Error Session::postHandshake() {
901 if (tx_pending_) {
902 Error rc = flush();
903 if (rc != Error::Ok) return rc;
904 }
905 const uint8_t *msg = nullptr;
906 size_t mlen = 0;
907 Error rc = readHandshake(msg, mlen);
908 if (rc != Error::Ok) return rc;
909
910 if (msg[0] == kHsNewSessionTicket) { // tickets are ignored
911 consumeHandshake();
912 return Error::Ok;
913 }
914 if (msg[0] != kHsKeyUpdate || mlen != 5 || msg[4] > 1) {
915 consumeHandshake();
916 return Error::UnexpectedMessage;
917 }
918 uint8_t request = msg[4];
919 consumeHandshake();
920
921 // Update the receive keys.
922 uint8_t next[32];
923 Hkdf hk;
924 if (!hk.expandLabel(rx_keys_.secret(), "traffic upd", nullptr, 0, next, 32)) return Error::Internal;
925 if (!rx_keys_.install(rx_keys_.cipher(), next)) return Error::Internal;
926
927 if (request) {
928 // Reply under the old sending keys, then update the sending keys.
929 static const uint8_t reply[5] = {kHsKeyUpdate, 0, 0, 1, 0};
930 rc = sendRecord(kTypeHandshake, reply, sizeof(reply));
931 if (rc != Error::Ok) return rc;
932 if (!hk.expandLabel(tx_keys_.secret(), "traffic upd", nullptr, 0, next, 32)) return Error::Internal;
933 if (!tx_keys_.install(tx_keys_.cipher(), next)) return Error::Internal;
934 return flush();
935 }
936 return Error::Ok;
937}

939inline int Session::write(const uint8_t *buf, size_t len) {
940 if (!handshake_complete_) return code(Error::BadInput);
941 if (tx_pending_) {
942 Error rc = flush();
943 if (rc != Error::Ok) return code(rc);
944 if (tx_user_len_) {
945 int n = (int)tx_user_len_;
946 tx_user_len_ = 0;
947 return n;
948 }
949 }
950 if (len == 0) return 0;
951 if (len > kMaxFragment) len = kMaxFragment;
952 Error rc = sendRecord(kTypeApplication, buf, len);
953 if (rc != Error::Ok) return code(rc);
954 tx_user_len_ = len;
955 rc = flush();
956 if (rc != Error::Ok) return code(rc);
957 tx_user_len_ = 0;
958 return (int)len;
959}

961inline int Session::read(uint8_t *buf, size_t len) {
962 if (!handshake_complete_) return code(Error::BadInput);
963 if (tx_pending_) {
964 Error rc = flush();
965 if (rc != Error::Ok) return code(rc);
966 }
967 if (len == 0) return 0;
968 for (;;) {
969 if (pt_len_ == 0) {
970 Error rc = fetchRecord();
971 if (rc != Error::Ok) return code(rc);
972 }
973 if (pt_type_ == kTypeApplication) {
974 size_t n = len < pt_len_ ? len : pt_len_;
975 memcpy(buf, rx_buf_.data() + pt_off_, n);
976 pt_off_ += n;
977 pt_len_ -= n;
978 return (int)n;
979 }
980 if (pt_type_ == kTypeHandshake) {
981 Error rc = postHandshake();
982 if (rc != Error::Ok) return code(rc);
983 continue;
984 }
985 return code(Error::UnexpectedMessage);
986 }
