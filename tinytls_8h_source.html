92 bool connect(const char *host, uint16_t port) {
93 stop();
94 session_->reset();
95 last_error_ = tinytls::Error::Ok;
96 if (transport_ == nullptr || !session_->setHostname(host)) {
97 last_error_ = tinytls::Error::BadInput;
98 return false;
99 }
100 session_->bind(transport_, &drbg_);
101 session_->configure(mode_, trust_, time_);
102 if (!transport_->connect(host, port)) {
103 last_error_ = tinytls::Error::ConnectionClosed;
104 return false;
105 }
106 uint32_t start = clock_ ? clock_() : 0;
107 tinytls::Error rc;
108 while ((rc = session_->handshake()) != tinytls::Error::Ok) {
109 if (rc != tinytls::Error::WantRead && rc != tinytls::Error::WantWrite) break;
110 if (timedOut(start)) {
111 rc = tinytls::Error::ConnectionClosed;
112 break;
113 }
114 idle();
115 }
116 if (rc != tinytls::Error::Ok) {
117 last_error_ = rc;
118 transport_->stop();
119 return false;
120 }
121 return true;
122 }
