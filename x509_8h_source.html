62class DerReader {
63 public:
65 DerReader(const uint8_t *data, size_t len) : p_(data), end_(data + len) {}
66
68 bool next(uint8_t &tag, const uint8_t *&val, size_t &len) {
69 if (p_ >= end_) return false;
70 uint8_t t = *p_;
71 if ((t & 0x1f) == 0x1f) return false; // multi-byte tags are not used in certificates
72 p_++;
73 if (p_ >= end_) return false;
74 uint8_t l = *p_++;
75 size_t n = 0;
76 if (l < 0x80) {
77 n = l;
78 } else {
79 int nb = l & 0x7f;
80 if (nb == 0 || nb > 4 || (size_t)(end_ - p_) < (size_t)nb) return false;
81 for (int i = 0; i < nb; i++) n = (n << 8) | *p_++;
82 }
83 if ((size_t)(end_ - p_) < n) return false;
84 tag = t;
85 val = p_;
86 len = n;
87 p_ += n;
88 return true;
89 }
90
92 bool expect(uint8_t tag, const uint8_t *&val, size_t &len) {
93 uint8_t t;
94 return next(t, val, len) && t == tag;
95 }
96
98 const uint8_t *position() const { return p_; }
99
101 bool atEnd() const { return p_ >= end_; }
102
103 private:
104 const uint8_t *p_;
105 const uint8_t *end_;
106};

115class Base64 {
116 public:
118 int decode(const uint8_t *in, size_t in_len, uint8_t *out, size_t out_cap) const {
119 size_t o = 0;
120 uint32_t acc = 0;
121 int bits = 0;
122 for (size_t i = 0; i < in_len; i++) {
123 uint8_t c = in[i];
124 int v;
125 if (c >= 'A' && c <= 'Z') v = c - 'A';
126 else if (c >= 'a' && c <= 'z') v = c - 'a' + 26;
127 else if (c >= '0' && c <= '9') v = c - '0' + 52;
128 else if (c == '+') v = 62;
129 else if (c == '/') v = 63;
130 else if (c == '=' || c == '\r' || c == '\n' || c == ' ' || c == '\t') continue;
131 else return -1;
132 acc = (acc << 6) | (uint32_t)v;
133 bits += 6;
134 if (bits >= 8) {
135 bits -= 8;
136 if (o >= out_cap) return -1;
137 out[o++] = (uint8_t)(acc >> bits);
138 }
139 }
140 return (int)o;
141 }

176 type_ = Type::None;
177 DerReader r(v, len);
178 const uint8_t *av, *bv;
179 size_t al, bl;
180 if (!r.expect(0x30, av, al)) return Error::X509BadCertificate;
181 DerReader alg(av, al);
182 const uint8_t *oid, *params_start = nullptr;
183 size_t oid_len;
184 if (!alg.expect(0x06, oid, oid_len)) return Error::X509BadCertificate;
185 params_start = alg.position();

187 if (!r.expect(0x03, bv, bl) || bl < 2 || bv[0] != 0) return Error::X509BadCertificate;
188 const uint8_t *key = bv + 1;
189 size_t key_len = bl - 1;
190
191 if (oidIs(oid, oid_len, detail::kEcPublicKey, sizeof(detail::kEcPublicKey))) {
192 DerReader pr(params_start, params_len);
193 const uint8_t *curve;
194 size_t curve_len;
195 if (!pr.expect(0x06, curve, curve_len) ||
196 !oidIs(curve, curve_len, detail::kPrime256v1, sizeof(detail::kPrime256v1))) {
197 return Error::X509Unsupported;
198 }
199 if (key_len != 65 || key[0] != 0x04) return Error::X509BadCertificate;
200 memcpy(x_, key + 1, 32);
201 memcpy(y_, key + 33, 32);
202 type_ = Type::Ec;
203 return Error::Ok;
204 }
205 if (oidIs(oid, oid_len, detail::kRsaEncryption, sizeof(detail::kRsaEncryption))) {
206 // RSAPublicKey ::= SEQUENCE { modulus INTEGER, publicExponent INTEGER }
207 DerReader seq(key, key_len);
208 const uint8_t *seqv;
209 size_t seql;
210 if (!seq.expect(0x30, seqv, seql) || !seq.atEnd()) return Error::X509BadCertificate;
211 DerReader ints(seqv, seql);
212 const uint8_t *nv, *ev;
213 size_t nl, el;
214 if (!ints.expect(0x02, nv, nl) || !ints.expect(0x02, ev, el)) return Error::X509BadCertificate;
215 magnitude(nv, nl, nv, nl);
216 magnitude(ev, el, ev, el);
217 if (!rsa_.set(nv, nl, ev, el)) return Error::X509Unsupported;
218 type_ = Type::Rsa;
219 return Error::Ok;
220 }
221 return Error::X509Unsupported;
222 }

229 Error verify(uint16_t scheme, const uint8_t *msg, size_t msg_len, const uint8_t *sig,
230 size_t sig_len) const {
231 uint8_t digest[32];
232 Sha256 h;
233 h.digest(msg, msg_len, digest);
234
235 if (scheme == 0x0403) {
236 if (type_ != Type::Ec) return Error::X509BadSignature;
237 uint8_t r[32] = {0}, s[32] = {0};
238 if (!derEcdsaSignature(sig, sig_len, r, s)) return Error::X509BadSignature;
239 std::unique_ptr<EcdsaP256> curve(new EcdsaP256());
240 return curve->verify(x_, y_, digest, 32, r, s) == Error::Ok ? Error::Ok : Error::X509BadSignature;
241 }
242 if (type_ != Type::Rsa) return Error::X509BadSignature;
243 if (scheme == 0x0401) {
244 return rsa_.verifyPkcs1Sha256(digest, sig, sig_len) == Error::Ok ? Error::Ok : Error::X509BadSignature;
245 }
246 if (scheme == 0x0804) {
247 return rsa_.verifyPssSha256(digest, sig, sig_len) == Error::Ok ? Error::Ok : Error::X509BadSignature;
248 }
249 return Error::X509BadSignature;
250 }

269 bool derEcdsaSignature(const uint8_t *sig, size_t sig_len, uint8_t r[32], uint8_t s[32]) const {
270 DerReader outer(sig, sig_len);
271 const uint8_t *seq;
272 size_t seql;

274 DerReader ints(seq, seql);
275 const uint8_t *rv, *sv;
276 size_t rl, sl;

284 memcpy(s + 32 - sml, sm, sml);

324 Error parse(const uint8_t *der, size_t len) {
325 reset();
326 der_ = der;
327 der_len_ = len;
328 DerReader top(der, len);
329 const uint8_t *v;
330 size_t vl;
331 if (!top.expect(0x30, v, vl) || !top.atEnd()) return Error::X509BadCertificate;
332
333 // Certificate ::= SEQUENCE { tbsCertificate, signatureAlgorithm, signatureValue }
334 DerReader cert(v, vl);
335 const uint8_t *tbs_start = cert.position();
336 if (!cert.expect(0x30, v, vl)) return Error::X509BadCertificate;
337 tbs_ = tbs_start;
338 tbs_len_ = (size_t)(cert.position() - tbs_start);
339
340 const uint8_t *alg_v;
341 size_t alg_l;
342 if (!cert.expect(0x30, alg_v, alg_l)) return Error::X509BadCertificate;
343 DerReader alg(alg_v, alg_l);
344 const uint8_t *oid;
345 size_t oid_len;
346 if (!alg.expect(0x06, oid, oid_len)) return Error::X509BadCertificate;
347 if (oidIs(oid, oid_len, detail::kEcdsaSha256, sizeof(detail::kEcdsaSha256))) {
348 alg_ = SignatureAlgorithm::EcdsaSha256;
349 } else if (oidIs(oid, oid_len, detail::kRsaSha256, sizeof(detail::kRsaSha256))) {
350 alg_ = SignatureAlgorithm::RsaPkcs1Sha256;
351 } else if (oidIs(oid, oid_len, detail::kRsaPss, sizeof(detail::kRsaPss))) {
352 // The PSS parameters must name SHA-256; its OID must appear in them.
353 bool sha256 = false;
354 for (const uint8_t *s = alg.position(); s + sizeof(detail::kSha256) <= alg_v + alg_l; s++) {
355 if (memcmp(s, detail::kSha256, sizeof(detail::kSha256)) == 0) sha256 = true;
356 }
357 if (!sha256) return Error::X509Unsupported;
358 alg_ = SignatureAlgorithm::RsaPssSha256;
359 } else {
360 return Error::X509Unsupported;
361 }
362
363 const uint8_t *sv;
364 size_t sl;
365 if (!cert.expect(0x03, sv, sl) || sl < 2 || sv[0] != 0 || !cert.atEnd()) {
366 return Error::X509BadCertificate;
367 }
368 sig_ = sv + 1;
369 sig_len_ = sl - 1;
370
371 return parseTbs(v, vl);
372 }

465 bool matchName(const uint8_t *pat, size_t pat_len, const char *host) const {
466 size_t host_len = strlen(host);
467 if (pat_len > 2 && pat[0] == '*' && pat[1] == '.') {
468 // "*.example.com" matches exactly one label in front of example.com.
469 const char *dot = strchr(host, '.');
470 if (dot == nullptr || dot == host) return false;
471 size_t suffix_len = pat_len - 1;
472 size_t label_len = (size_t)(dot - host);
473 if (host_len - label_len != suffix_len) return false;
474 return ciEqual(pat + 1, suffix_len, dot, suffix_len);
475 }
476 return ciEqual(pat, pat_len, host, host_len);
477 }

479 bool ciEqual(const uint8_t *a, size_t al, const char *b, size_t bl) const {
480 if (al != bl) return false;
481 for (size_t i = 0; i < al; i++) {
482 uint8_t x = a[i], y = (uint8_t)b[i];
483 if (x >= 'A' && x <= 'Z') x = (uint8_t)(x + 32);
484 if (y >= 'A' && y <= 'Z') y = (uint8_t)(y + 32);
485 if (x != y) return false;
486 }
487 return true;
488 }

496 int64_t daysFromCivil(int64_t y, unsigned m, unsigned d) const {
497 y -= m <= 2;
498 int64_t era = (y >= 0 ? y : y - 399) / 400;
499 int64_t yoe = y - era * 400;
500 int64_t doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
501 int64_t doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
502 return era * 146097 + doe - 719468;
503 }

506 bool parseTime(uint8_t tag, const uint8_t *v, size_t len, int64_t &out) const {
507 int year, off;
508 if (tag == 0x17 && len == 13) {
509 int yy = twoDigits(v);
510 if (yy < 0) return false;
511 year = yy < 50 ? 2000 + yy : 1900 + yy;
512 off = 2;
513 } else if (tag == 0x18 && len == 15) {
514 int hi = twoDigits(v), lo = twoDigits(v + 2);
515 if (hi < 0 || lo < 0) return false;
516 year = hi * 100 + lo;
517 off = 4;
518 } else {
519 return false;
520 }
521 if (v[len - 1] != 'Z') return false;
522 int mon = twoDigits(v + off), day = twoDigits(v + off + 2);
523 int hour = twoDigits(v + off + 4), min = twoDigits(v + off + 6), sec = twoDigits(v + off + 8);
524 if (mon < 1 || mon > 12 || day < 1 || day > 31 || hour < 0 || hour > 23 || min < 0 ||
525 min > 59 || sec < 0 || sec > 60) {
526 return false;
527 }
528 out = daysFromCivil(year, (unsigned)mon, (unsigned)day) * 86400 + hour * 3600 + min * 60 + sec;
529 return true;
530 }

532 Error parseTbs(const uint8_t *v, size_t vl) {
533 DerReader t(v, vl);
534 const uint8_t *e;
535 size_t el;
536 uint8_t tag;
537 const uint8_t *cv;
538 size_t cl;
539 // Optional explicit version [0], then serialNumber and the inner signature algorithm.
540 if (!t.atEnd() && *t.position() == 0xa0 && !t.next(tag, cv, cl)) return Error::X509BadCertificate;
541 if (!t.expect(0x02, e, el)) return Error::X509BadCertificate;
542 if (!t.expect(0x30, e, el)) return Error::X509BadCertificate;
543
544 const uint8_t *issuer_start = t.position();
545 if (!t.expect(0x30, e, el)) return Error::X509BadCertificate;
546 issuer_ = issuer_start;
547 issuer_len_ = (size_t)(t.position() - issuer_start);
548
549 if (!t.expect(0x30, e, el)) return Error::X509BadCertificate;
550 {
551 DerReader times(e, el);
552 const uint8_t *a, *b;
553 size_t al, bl;
554 uint8_t ta, tb;
555 if (!times.next(ta, a, al) || !times.next(tb, b, bl)) return Error::X509BadCertificate;
556 if (!parseTime(ta, a, al, not_before_) || !parseTime(tb, b, bl, not_after_)) {
557 return Error::X509BadCertificate;
558 }
559 }
560
561 const uint8_t *subject_start = t.position();
562 if (!t.expect(0x30, e, el)) return Error::X509BadCertificate;
563 subject_ = subject_start;
564 subject_len_ = (size_t)(t.position() - subject_start);
565
566 if (!t.expect(0x30, e, el)) return Error::X509BadCertificate;
567 if (key_.parse(e, el) != Error::Ok) return Error::X509Unsupported;
568
569 // Optional unique IDs and extensions [3].
570 while (!t.atEnd()) {
571 if (!t.next(tag, e, el)) return Error::X509BadCertificate;
572 if (tag != 0xa3) continue;
573 DerReader ext_list(e, el);
574 const uint8_t *list;
575 size_t list_len;
576 if (!ext_list.expect(0x30, list, list_len)) return Error::X509BadCertificate;
577 DerReader exts(list, list_len);
578 while (!exts.atEnd()) {
579 const uint8_t *ext;
580 size_t ext_len;
581 if (!exts.expect(0x30, ext, ext_len)) return Error::X509BadCertificate;
582 DerReader fields(ext, ext_len);
583 const uint8_t *oid;
584 size_t oid_len;
585 const uint8_t *fv;
586 size_t fl;
587 uint8_t ft;
588 if (!fields.expect(0x06, oid, oid_len)) return Error::X509BadCertificate;
589 if (!fields.next(ft, fv, fl)) return Error::X509BadCertificate;
590 if (ft == 0x01) { // critical flag
591 if (!fields.next(ft, fv, fl)) return Error::X509BadCertificate;
592 }
593 if (ft != 0x04) return Error::X509BadCertificate; // extnValue OCTET STRING
594
595 if (oidIs(oid, oid_len, detail::kBasicConstraints, sizeof(detail::kBasicConstraints))) {
596 DerReader bc(fv, fl);
597 const uint8_t *bv;
598 size_t bl;
599 if (!bc.expect(0x30, bv, bl)) return Error::X509BadCertificate;
600 has_basic_constraints_ = true;
601 is_ca_ = false;
602 DerReader body(bv, bl);
603 uint8_t bt;
604 const uint8_t *bbv;
605 size_t bbl;
606 if (!body.atEnd() && body.next(bt, bbv, bbl) && bt == 0x01 && bbl == 1) {
607 is_ca_ = bbv[0] != 0;
608 }
609 } else if (oidIs(oid, oid_len, detail::kSubjectAltName, sizeof(detail::kSubjectAltName))) {
610 DerReader san(fv, fl);
611 if (!san.expect(0x30, san_, san_len_)) return Error::X509BadCertificate;
612 }
613 }
614 }
615 return Error::Ok;
616 }

689 Error addPem(const uint8_t *buf, size_t len) {
690 static const char begin[] = "-----BEGIN CERTIFICATE-----";
691 static const char end[] = "-----END CERTIFICATE-----";
692 const size_t bl = sizeof(begin) - 1, el = sizeof(end) - 1;
693 size_t added = 0, pos = 0;
694 while (pos < len) {
695 const uint8_t *body = nullptr;
696 for (size_t i = pos; i + bl <= len; i++) {
697 if (memcmp(buf + i, begin, bl) == 0) {
698 body = buf + i + bl;
699 break;
700 }

707 break;
708 }

714 Error rc = addDer(der.data(), (size_t)n);
715 if (rc != Error::Ok) return rc;
716 added++;
717 pos = (size_t)(body_end - buf) + el;
718 }
719 return added > 0 ? Error::Ok : Error::X509BadCertificate;
720 }

726 return addPem(buf, len);
727 }
728 return addDer(buf, len);
729 }

756 public:
762 ChainVerifier(const CertificateStore *trust, const char *host, int64_t now)
763 : trust_(trust), host_(host), now_(now) {}
764
775 Error verify(const Certificate *chain, size_t n, uint32_t &flags) const {
776 uint32_t f = 0;
777 if (n == 0) {
778 flags |= VerifyBadCertificate;
779 return Error::X509BadCertificate;
780 }
781 if (host_ && !chain[0].matchesHost(host_)) f |= VerifyNameMismatch;
782
783 if (now_ < 0) {
784 f |= VerifyTimeUnchecked;
785 } else {
786 for (size_t i = 0; i < n; i++) {
787 if (now_ < chain[i].notBefore()) f |= VerifyFuture;
788 if (now_ > chain[i].notAfter()) f |= VerifyExpired;
789 }
790 }
791
792 for (size_t i = 0; i + 1 < n; i++) {
793 const Certificate &issuer = chain[i + 1];
794 if (!issuer.issued(chain[i])) f |= VerifyNotTrusted;
795 if (!issuer.isCa()) f |= VerifyNotTrusted;
796 if (issuer.verifySignatureOf(chain[i]) != Error::Ok) f |= VerifyBadSignature;
797 }
798
799 const Certificate &last = chain[n - 1];
800 bool trusted = false;
801 if (trust_) {
802 for (size_t t = 0; t < trust_->size() && !trusted; t++) {
803 const Certificate &anchor = trust_->at(t);
804 if (anchor.sameAs(last)) {
805 trusted = true; // the chain includes the anchor itself
806 break;
807 }
808 if (!anchor.issued(last) || !anchor.isCa()) continue;
809 if (now_ >= 0 && (now_ < anchor.notBefore() || now_ > anchor.notAfter())) continue;
810 if (anchor.verifySignatureOf(last) == Error::Ok) trusted = true;
811 }
812 }
813 if (!trusted) f |= VerifyNotTrusted;
814
815 flags |= f;
816 return f == 0 ? Error::Ok : Error::X509CertVerifyFailed;
817 }
818
819 private:
820 const CertificateStore *trust_;
821 const char *host_;
822 int64_t now_;
823};
